# requirements.txt — Mycorrhizal Barcelona / NEXUS-Micro pipeline
# Python 3.13 compatible. Tested against pandas 3.0.2 + numpy 2.4.3 (already installed).
# Install: pip install -r requirements.txt

matplotlib>=3.10.9         # static plots and map figure export

# ── Progress / utilities ──────────────────────────────────────────────────────
tqdm>=4.67.3               # download and processing progress bars

# ── Jupyter ───────────────────────────────────────────────────────────────────
